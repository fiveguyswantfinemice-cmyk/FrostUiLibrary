Load Library

local Rayfield = https://raw.githubusercontent.com/fiveguyswantfinemice-cmyk/FrostUiLibrary/refs/heads/main/source.lua

Window Setup

local Window = Rayfield:CreateWindow({
    Name = "Aether Hub V2",
    LoadingTitle = "Aether Suite",
    LoadingSubtitle = "by Developer"
})

Tabs

local CombatTab = Window:CreateTab("Combat", 4483345998)
local MovementTab = Window:CreateTab("Movement", 4483345998)

Buttons

MovementTab:CreateButton({
    Name = "Instant Teleport to Safezone",
    Callback = function()
        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = CFrame.new(0, 50, 0)
    end,
})

Toggles

CombatTab:CreateToggle({
    Name = "Enable Target Lock",
    Default = false,
    Flag = "TargetLockFlag",
    Callback = function(State)
        print("Target Lock active:", State)
    end,
})

Sliders

MovementTab:CreateSlider({
    Name = "Sprint Speed Modifier",
    Range = {16, 250},
    Increment = 1,
    CurrentValue = 16,
    Callback = function(Value)
        game.Players.LocalPlayer.Character.Humanoid.WalkSpeed = Value
    end,
})

Keybinds

CombatTab:CreateKeybind({
    Name = "Quick Target Key",
    CurrentKeybind = "E",
    HoldToInteract = false,
    Callback = function(Key)
        print("Quick key pressed!")
    end,
})
